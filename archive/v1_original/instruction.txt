Python

from datasets import load_dataset, concatenate_datasets, Dataset

def unwrap_empathy(example):
    return {"prompt": example["context"], "response": example["utterance"]}

def unwrap_crisis(example):
    return {"prompt": example["question"], "response": example["answer"]}

emp  = load_dataset("empathetic_dialogues", split="train").shuffle(seed=42).select(range(15000)).map(unwrap_empathy, remove_columns=emp.column_names)
mhc  = load_dataset("mental_health_conversations", split="train").shuffle(seed=42).select(range(15000)).map(lambda x: {"prompt": x["Context"], "response": x["Response"]})
cris = load_dataset("crisis_counseling_60k", split="train").shuffle(seed=42).select(range(12000)).map(unwrap_crisis)

train = concatenate_datasets([emp, mhc, cris]).shuffle(seed=42)
train.to_json("empathy_42k.jsonl")

3.3 Training script (rank-16, 1 epoch, 4-bit)
Save as train_lora.py:
Python
Copy

import torch, json, os
from transformers import AutoModelForCausalLM, AutoTokenizer
from peft import LoraConfig, get_peft_model, TaskType
from datasets import load_dataset

base = "meta-llama/Llama-3.1-8B-Instruct"   # needs HuggingFace auth
tokenizer = AutoTokenizer.from_pretrained(base, use_fast=True)
model = AutoModelForCausalLM.from_pretrained(
    base,
    load_in_4bit=True,
    device_map="auto",
    torch_dtype=torch.bfloat16,
    use_cache=False,
)

lora_config = LoraConfig(
    r=16,
    lora_alpha=32,
    target_modules=["q_proj", "v_proj", "k_proj", "o_proj", "gate_proj", "up_proj", "down_proj"],
    lora_dropout=0.05,
    bias="none",
    task_type=TaskType.CAUSAL_LM,
)

model = get_peft_model(model, lora_config)

def format_chat(sample):
    chat = [
        {"role": "system", "content": "You are Relevia, 26, empathetic friend, survivor of anxiety & grief. Reply in 2-4 warm sentences, one grounding tool, no medical claims."},
        {"role": "user", "content": sample["prompt"]},
        {"role": "assistant", "content": sample["response"]},
    ]
    return {"text": tokenizer.apply_chat_template(chat, tokenize=False)}

data = load_dataset("json", data_files="empathy_42k.jsonl", split="train").map(format_chat)

from transformers import Trainer, TrainingArguments
trainer = Trainer(
    model=model,
    train_dataset=data,
    args=TrainingArguments(
        output_dir="./lora_out",
        per_device_train_batch_size=1,
        gradient_accumulation_steps=16,
        num_train_epochs=1,
        learning_rate=2e-4,
        fp16=False,
        bf16=True,
        logging_steps=50,
        save_strategy="epoch",
        report_to=None,
    ),
    data_collator=lambda x: tokenizer([d["text"] for d in x], return_tensors="pt", padding=True, truncation=True, max_length=2048),
)
trainer.train()
model.save_pretrained("./lora_out")
